local function FarmIsPlayersNear(Distance)
	local LocalPlayer = Players.LocalPlayer
	local LocalRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if LocalRoot then
		for _, v in Players:GetPlayers() do
			if v ~= LocalPlayer and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
				if (v.Character.HumanoidRootPart.Position - LocalRoot.Position).Magnitude < Distance then
					return true
				end
			end
		end
	end
	return false
end

local function FarmGetGenerators()
	local t16 = {}
	local InGame = workspace:FindFirstChild("Map") and workspace.Map:FindFirstChild("Ingame")
	if InGame then
		local GameMap = InGame:FindFirstChild("Map")
		if GameMap then
			for _, child in pairs(GameMap:GetChildren()) do
				if child.Name == "Generator" and child:FindFirstChild("Main") then
					table.insert(t16, child)
				end
			end
		end
	end
	return t16
end

local function FarmGetProgress(Generator)
	local Progress = Generator:FindFirstChild("Progress")
	if Progress and Progress:IsA("NumberValue") then return Progress.Value end
	return 0
end

local function FarmCompleteGenerator(Generator)
	local Remotes = Generator:FindFirstChild("Remotes")
	local Progress = Generator:FindFirstChild("Progress")
	if not (Remotes and Progress) then return end
	local RemoteEvent = Remotes:FindFirstChildOfClass("RemoteEvent")
	if not RemoteEvent then return end
	RemoteEvent:FireServer()
end

local function FarmGetGeneratorPos(Generator)
	local Instances = Generator:FindFirstChild("Instances")
	if Instances then
		local Cube = Instances:FindFirstChild("Cube.001")
		if Cube and Cube:IsA("BasePart") then return Cube.Position end
	end
	local Positions = Generator:FindFirstChild("Positions")
	if Positions then
		local Center = Positions:FindFirstChild("Center")
		if Center and Center:IsA("BasePart") then return Center.Position end
	end
	local Main = Generator:FindFirstChild("Main")
	if Main and Main:IsA("BasePart") then return Main.Position end
	for _, d in pairs(Generator:GetDescendants()) do
		if d:IsA("BasePart") then return d.Position end
	end
	return Vector3.new(0,0,0)
end

local function FarmGetNearestGenerator()
	local localChar = Players.LocalPlayer.Character
	local root = localChar and localChar:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local best, bestDist = nil, math.huge
	for _, gen in pairs(FarmGetGenerators()) do
		if FarmGetProgress(gen) < 100 then
			local pos = FarmGetGeneratorPos(gen)
			if pos ~= Vector3.new(0,0,0) then
				local d = (pos - root.Position).Magnitude
				if d < bestDist then bestDist = d best = gen end
			end
		end
	end
	return best
end

local function FarmFirePrompt(Generator)
	local Main = Generator:FindFirstChild("Main")
	if not Main then return false end
	local prompt = Main:FindFirstChild("Prompt")
	if not prompt then
		for _, d in pairs(Main:GetDescendants()) do
			if d:IsA("ProximityPrompt") then prompt = d break end
		end
	end
	if prompt and prompt:IsA("ProximityPrompt") then
		(fireproximityprompt or getgenv().fireproximityprompt)(prompt)
		return true
	end
	return false
end

local function FarmFireGeneratorRemote(Generator)
	local Remotes = Generator:FindFirstChild("Remotes")
	if Remotes then
		local RE = Remotes:FindFirstChild("RE")
		if RE and RE:IsA("RemoteEvent") then
			local ok = pcall(function() RE:FireServer() end)
			return ok
		end
	end
	return false
end

local function uFarmStart()
	if t1.autoFarmActive then return end
	t1.autoFarmActive = true

	t4.autoFarmConnection = RunService.Heartbeat:Connect(function()
		if not t1.autoFarmActive then return end
		local LocalPlayer = Players.LocalPlayer
		local LocalCharacter = LocalPlayer.Character
		local LocalRoot = LocalCharacter and LocalCharacter:FindFirstChild("HumanoidRootPart")
		if not LocalRoot then return end

		for _, Generator in FarmGetGenerators() do
			local Main = Generator:FindFirstChild("Main")
			if not Main then continue end
			if (Main.Position - LocalRoot.Position).Magnitude < 6.7 then
				if not u5.currentFarmGenerator then
					u5.currentFarmGenerator = Generator
					u5.farmFixActive = true
					task.spawn(function()
						while t1.autoFarmActive do
							if not u5.currentFarmGenerator or not u5.currentFarmGenerator.Parent then break end
							if FarmGetProgress(u5.currentFarmGenerator) >= 100 then break end
							if (u5.currentFarmGenerator:FindFirstChild("Main").Position - LocalRoot.Position).Magnitude > 8 then break end
							local isNear = FarmIsPlayersNear(t3.farmNearDistance)
							local baseCD = t3.farmCooldown
							local GenCD = (not isNear and t3.farmSpeedUpNoNear and t3.farmMinCooldown) or
								(baseCD + (math.random() - 0.5) * 0.2 * baseCD)
							task.wait(GenCD)
							if not t1.autoFarmActive then break end
							if not u5.currentFarmGenerator or not u5.currentFarmGenerator.Parent then break end
							FarmCompleteGenerator(u5.currentFarmGenerator)
						end
						u5.farmFixActive = false
						u5.currentFarmGenerator = nil
					end)
				end
				break
			end
		end
	end)
end

local function uFarmStop()
	t1.autoFarmActive = false
	u5.currentFarmGenerator = nil
	u5.farmFixActive = false
	if t4.autoFarmConnection then
		t4.autoFarmConnection:Disconnect()
		t4.autoFarmConnection = nil
	end
end

	end)

local function BlockCheckCooldown()
	if tick() - t2.lastBlockFire >= t3.blockCooldown then
		t2.lastBlockFire = tick()
		return true
	end
	return false
end

local function BlockFire()
	local RemoteEvent = game.ReplicatedStorage.Modules.Network.RemoteEvent
	pcall(function() RemoteEvent:FireServer("UseActorAbility", "Block") end)
end

local function BlockMethodAnimation()
	if not t1.blockOnAnimation then return end
	if not u5.isSurvivor or not u5.killerModel then return end
	local Character = Players.LocalPlayer.Character
	if not Character or not Character:FindFirstChild("HumanoidRootPart") then return end
	if not u5.killerModel:FindFirstChild("HumanoidRootPart") then return end
	if (Character.HumanoidRootPart.Position - u5.killerModel.HumanoidRootPart.Position).Magnitude > t3.blockRange then return end
	local Humanoid = u5.killerModel:FindFirstChildOfClass("Humanoid")
	if not Humanoid then return end
	local Animator = Humanoid:FindFirstChildOfClass("Animator")
	if not Animator then return end
	for _, v in pairs(Animator:GetPlayingAnimationTracks()) do
		if v.Animation and table.find(t3.blockAnimationIds, v.Animation.AnimationId) and u31("Block") then
			if BlockCheckCooldown() then BlockFire() end
			return
		end
	end
end

local function BlockMethodHitbox()
	if not t1.blockOnHitbox then return end
	if not u5.isSurvivor or not u5.killerModel then return end
	local Character = Players.LocalPlayer.Character
	if not Character then return end
	local SelfHitbox = Character:FindFirstChild("QueryHitbox")
	local KillerHitbox = u5.killerModel:FindFirstChild("QueryHitbox")
	if not SelfHitbox or not KillerHitbox then return end
	local Params = OverlapParams.new()
	Params.FilterType = Enum.RaycastFilterType.Include
	Params.MaxParts = 1
	Params.FilterDescendantsInstances = { SelfHitbox }
	local Part = Instance.new("Part")
	Part.Name = "SNT_BlockDetect"
	Part.Size = Vector3.new(5.2, 6, 5.65) * 2.2
	Part.CFrame = KillerHitbox.CFrame * CFrame.new(0, 0, -3.5)
	Part.CanCollide = false
	Part.Anchored = true
	Part.CastShadow = false
	Part.Transparency = 1
	Part.Parent = workspace
	local Hit = workspace:GetPartsInPart(Part, Params)
	Part:Destroy()
	if #Hit > 0 and u31("Block") then
		if BlockCheckCooldown() then BlockFire() end
	end
end

local function BlockMethodDistance()
	if not t1.blockOnDistance then return end
	if not u5.isSurvivor or not u5.killerModel then return end
	local Character = Players.LocalPlayer.Character
	if not Character or not Character:FindFirstChild("HumanoidRootPart") then return end
	if not u5.killerModel:FindFirstChild("HumanoidRootPart") then return end
	local dist = (Character.HumanoidRootPart.Position - u5.killerModel.HumanoidRootPart.Position).Magnitude
	if dist <= 8 and u31("Block") then
		if BlockCheckCooldown() then BlockFire() end
	end
end

local function BlockMethodProximityScan()
	if not u5.isSurvivor or not u5.killerModel then return end
	local Character = Players.LocalPlayer.Character
	if not Character or not Character:FindFirstChild("HumanoidRootPart") then return end
	if not u5.killerModel:FindFirstChild("HumanoidRootPart") then return end
	if (Character.HumanoidRootPart.Position - u5.killerModel.HumanoidRootPart.Position).Magnitude > 12 then return end
	local Humanoid = u5.killerModel:FindFirstChildOfClass("Humanoid")
	if not Humanoid then return end
	local Animator = Humanoid:FindFirstChildOfClass("Animator")
	if not Animator then return end
	local tracks = Animator:GetPlayingAnimationTracks()
	if #tracks > 0 then
		for _, track in pairs(tracks) do
			local id = track.Animation and track.Animation.AnimationId
			if id and (id:find("slash") or id:find("Slash") or id:find("attack") or id:find("Attack") or id:find("swing") or id:find("Swing")) then
				if u31("Block") and BlockCheckCooldown() then BlockFire() end
				return
			end
		end
	end
end

local function uBlockStart()
	if t1.autoBlockNewActive then return end
	t1.autoBlockNewActive = true
	t4.autoBlockNewConnection = RunService.Heartbeat:Connect(function()
		if not t1.autoBlockNewActive then return end

		elseif t1.autoBlockMethod == 3 then BlockMethodDistance()
		elseif t1.autoBlockMethod == 4 then BlockMethodProximityScan() end
	end)
end

local function uBlockStop()
	t1.autoBlockNewActive = false
	if t4.autoBlockNewConnection then t4.autoBlockNewConnection:Disconnect() t4.autoBlockNewConnection = nil end
end

local function CreateBillboard(p13, p14, p15, p16)
	if not p13 or not p13:IsA("BasePart") then return nil end
	local BillboardGui = p13:FindFirstChild("SNT_Billboard")
	if BillboardGui then
		local lbl = BillboardGui:FindFirstChild("TextLabel")
		if lbl then
			lbl.Text = p14 or ""
			lbl.TextColor3 = p15 or Color3.fromRGB(255,255,255)
		end
		return BillboardGui
	end
	BillboardGui = Instance.new("BillboardGui")
	BillboardGui.Name = "SNT_Billboard"
	BillboardGui.Adornee = p13
	BillboardGui.Size = UDim2.new(0, 200, 0, 50)
	BillboardGui.StudsOffset = p16 or Vector3.new(0, 3, 0)
	BillboardGui.AlwaysOnTop = true
	BillboardGui.MaxDistance = t3.espMaxDistance
	BillboardGui.Parent = p13
	local TextLabel = Instance.new("TextLabel")
	TextLabel.Name = "TextLabel"
	TextLabel.Size = UDim2.new(1, 0, 1, 0)
	TextLabel.BackgroundTransparency = 1
	TextLabel.Text = p14 or ""
	TextLabel.TextColor3 = p15 or Color3.fromRGB(255,255,255)
	TextLabel.TextSize = t3.espTextSize or 14
	TextLabel.Font = Enum.Font.GothamBold
	TextLabel.TextStrokeTransparency = 0
	TextLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
	TextLabel.Parent = BillboardGui
	return BillboardGui
end

local function CreateHighlight(p17, p18)
	if not p17 or not p17:IsA("Model") then return nil end
	local Highlight = p17:FindFirstChild("SNT_Highlight")
	if Highlight then
		Highlight.FillColor = p18 or Color3.fromRGB(255,255,255)
		Highlight.OutlineColor = p18 or Color3.fromRGB(255,255,255)
		Highlight.FillTransparency = t3.highlightTransparency or 0.5
		Highlight.OutlineTransparency = t3.highlightOutlineTransparency or 0

local function RefreshSurvivors()
	local map = espState.survivorMap
	local seen = {}
	local Players2 = workspace:FindFirstChild("Players")
	local Survivors = Players2 and Players2:FindFirstChild("Survivors")
	if Survivors then
		for _, child in pairs(Survivors:GetChildren()) do
			if child:IsA("Model") then
				seen[child] = true
				local username = child:GetAttribute("Username") or child.Name
				if u8.SurvivorsHighlight then
					local hl = map[child]
					if not hl then
						hl = { highlight = CreateHighlight(child, t11.Survivor), billboard = nil, head = nil }
						map[child] = hl
					end
					if hl.highlight then
						hl.highlight.FillTransparency = t3.highlightTransparency or 0.5
						hl.highlight.OutlineTransparency = t3.highlightOutlineTransparency or 0
					end
				elseif map[child] and map[child].highlight then
					map[child].highlight:Destroy()
					map[child].highlight = nil
				end
				if u8.SurvivorsNameESP or u8.SurvivorsDistance then
					local Head = child:FindFirstChild("Head")
					if Head then
						local entry = map[child]
						if not entry then entry = { highlight = nil, billboard = nil, head = Head } map[child] = entry end
						entry.head = Head
						local lblText = (u8.SurvivorsNameESP and username or "")
						local bb = entry.billboard
						if not bb or not bb.Parent then
							bb = CreateBillboard(Head, lblText, t11.Survivor, Vector3.new(0,4,0))
							entry.billboard = bb
						else
							local l = bb:FindFirstChild("TextLabel")
							if l and l.Text ~= lblText and not u8.SurvivorsDistance then l.Text = lblText end
						end
					end
				elseif map[child] and map[child].billboard then
					map[child].billboard:Destroy()
					map[child].billboard = nil
				end
			end
		end
	end
	for inst, entry in pairs(map) do
		if not seen[inst] then
			if entry.highlight then entry.highlight:Destroy() end
			if entry.billboard then entry.billboard:Destroy() end
			map[inst] = nil
		end
	end
end

local function ScanItems()
	local list = {}
	local function recurse(node)
		for _, child in pairs(node:GetChildren()) do
			if child:IsA("Tool") then
				table.insert(list, child)
			elseif child:IsA("Folder") or child:IsA("Model") then
				recurse(child)
			end
		end
	end
	if workspace then recurse(workspace) end
	return list
end

local function RefreshItems(force)
	local now = tick()
	if not force and now - espState.lastItemScan < t3.espItemScanInterval then return end
	espState.lastItemScan = now

	local map = espState.itemMap
	local seen = {}

	for inst, entry in pairs(map) do
		if not inst.Parent then
			if entry.highlight then entry.highlight:Destroy() end
			if entry.billboard then entry.billboard:Destroy() end
			map[inst] = nil
		end
	end

	if u8.ItemsHighlight or u8.ItemsNameESP or u8.ItemsDistance then
		local list = ScanItems()
		for _, tool in pairs(list) do
			local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildWhichIsA("BasePart")
			if handle then
				seen[tool] = true
				local entry = map[tool]
				if not entry then entry = { highlight = nil, billboard = nil, handle = handle } map[tool] = entry end
				entry.handle = handle
				if u8.ItemsHighlight then
					if not entry.highlight then entry.highlight = CreateHighlight(tool, t11.Items) end
					if entry.highlight then
						entry.highlight.FillTransparency = t3.highlightTransparency or 0.5
						entry.highlight.OutlineTransparency = t3.highlightOutlineTransparency or 0
					end
				elseif entry.highlight then
					entry.highlight:Destroy()
					entry.highlight = nil
				end
				if u8.ItemsNameESP or u8.ItemsDistance then
					local lblText = (u8.ItemsNameESP and tool.Name or "")
					if not entry.billboard or not entry.billboard.Parent then
						entry.billboard = CreateBillboard(handle, lblText, t11.Items, Vector3.new(0,2,0))
					elseif not u8.ItemsDistance then
						local l = entry.billboard:FindFirstChild("TextLabel")
						if l and l.Text ~= lblText then l.Text = lblText end
					end
				elseif entry.billboard then
					entry.billboard:Destroy()
					entry.billboard = nil
				end
			end
		end
	end

	for inst, entry in pairs(map) do
		if not seen[inst] then
			if entry.highlight then entry.highlight:Destroy() end
			if entry.billboard then entry.billboard:Destroy() end
			map[inst] = nil
		end
	end
end

local function ScanGenerators()

			elseif child:IsA("Folder") or child:IsA("Model") then
				recurse(child)
			end

	if workspace then recurse(workspace) end
	return list
end

local distanceConnection = nil
local function StartDistanceUpdater()
	if distanceConnection then return end
	distanceConnection = RunService.Heartbeat:Connect(function()
		local now = tick()
		if now - espState.lastDistanceUpdate < t3.espDistanceUpdateInterval then return end
		espState.lastDistanceUpdate = now
		local root = GetLocalRoot()
		if not root then return end
		local rp = root.Position
		local maxD = t3.espMaxDistance

		if u8.SurvivorsDistance or u8.SurvivorsNameESP then
			for inst, entry in pairs(espState.survivorMap) do
				if entry.billboard and entry.billboard.Parent and entry.head and entry.head.Parent then
					local d = math.floor((entry.head.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end
						else
							local uname = inst:GetAttribute("Username") or inst.Name
							local base = u8.SurvivorsNameESP and uname or ""
							local txt = u8.SurvivorsDistance and (base .. " [" .. d .. "m]") or base
							if l.Text ~= txt then l.Text = txt end
							if l.TextTransparency ~= 0 then l.TextTransparency = 0 end
						end
					end
				end
			end
		end
		if u8.KillerDistance or u8.KillerNameESP then
			for inst, entry in pairs(espState.killerMap) do
				if entry.billboard and entry.billboard.Parent and entry.head and entry.head.Parent then
					local d = math.floor((entry.head.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end
						else
							local uname = inst:GetAttribute("Username") or inst.Name
							local base = u8.KillerNameESP and uname or ""
							local txt = u8.KillerDistance and (base .. " [" .. d .. "m]") or base
							if l.Text ~= txt then l.Text = txt end
							if l.TextTransparency ~= 0 then l.TextTransparency = 0 end
						end
					end
				end
			end
		end
		if u8.ItemsDistance or u8.ItemsNameESP then
			for inst, entry in pairs(espState.itemMap) do
				if entry.billboard and entry.billboard.Parent and entry.handle and entry.handle.Parent then
					local d = math.floor((entry.handle.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end
						else
							local base = u8.ItemsNameESP and inst.Name or ""
							local txt = u8.ItemsDistance and (base .. " [" .. d .. "m]") or base
							if l.Text ~= txt then l.Text = txt end
							if l.TextTransparency ~= 0 then l.TextTransparency = 0 end
						end
					end
				end
			end
		end
		if u8.GeneratorsDistance or u8.GeneratorsNameESP then
			for inst, entry in pairs(espState.generatorMap) do
				if entry.billboard and entry.billboard.Parent and entry.part and entry.part.Parent then
					local d = math.floor((entry.part.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end
						else
							local base = u8.GeneratorsNameESP and "Generator" or ""
							local txt = u8.GeneratorsDistance and (base .. " [" .. d .. "m]") or base
							if l.Text ~= txt then l.Text = txt end
							if l.TextTransparency ~= 0 then l.TextTransparency = 0 end
						end
					end
				end
			end
		end
	end)
end

local function u62(force)
	pcall(function()
		RefreshSurvivors()
		RefreshKillers()
		RefreshItems(force)
		RefreshGenerators(force)
	end)
end

function cleanupAllEsp()
	for _, map in pairs(espState) do
		if type(map) == "table" then
			for inst, entry in pairs(map) do
				if type(entry) == "table" then
					if entry.highlight then pcall(function() entry.highlight:Destroy() end) end
					if entry.billboard then pcall(function() entry.billboard:Destroy() end) end
				end
				map[inst] = nil
			end
		end
	end
	if distanceConnection then
		distanceConnection:Disconnect()
		distanceConnection = nil
	end
end

			u62(false)
		end
	end)
end

local m4Sk8Saved = nil

local function m4Sk8GetHum()
	local c = m4LP.Character

local function m4Sk8Save(h)
	if not h or m4Sk8Saved then return end
	local t = {}
	pcall(function()
		t.WalkSpeed = h.WalkSpeed
		pcall(function() t.JumpPower = h.JumpPower end)
		pcall(function() t.JumpPower = h.JumpHeight end)
		local ok, r = pcall(function() return h.AutoRotate end)
		if ok then t.AutoRotate = r end
		t.PlatformStand = h.PlatformStand
	end)
	m4Sk8Saved = t
end
local function m4Sk8Restore(h)
	if not h or not m4Sk8Saved then return end
	local s = m4Sk8Saved
	pcall(function()
		if s.WalkSpeed ~= nil then h.WalkSpeed = s.WalkSpeed end
		if s.JumpPower ~= nil and not pcall(function() h.JumpPower = s.JumpPower end) then
			pcall(function() h.JumpHeight = s.JumpPower end)
		end
		if s.AutoRotate ~= nil then pcall(function() h.AutoRotate = s.AutoRotate end) end
		if s.PlatformStand ~= nil then h.PlatformStand = s.PlatformStand end
	end)
	m4Sk8Saved = nil
end
local function m4Sk8Playing()
	local h = m4Sk8GetHum()
	if not h then return false end
	for _, tr in ipairs(h:GetPlayingAnimationTracks()) do
		local ok, id = pcall(function()
			return tostring(tr.Animation and tr.Animation.AnimationId or ""):match("%d+")
		end)
		if ok and id and table.find(m4Sk8AnimIds, id) then return true end
	end
	return false
end
local function m4Sk8On()
	if t1.m4Sk8Control then return end
	local h = m4Sk8GetHum()
	if not h then return end
	t1.m4Sk8Control = true

	pcall(function() h.WalkSpeed = 60 h.AutoRotate = false end)
	m4UIS.MouseBehavior = Enum.MouseBehavior.LockCenter

		local c = m4LP.Character

			r.CFrame = CFrame.new(r.Position, Vector3.new(cam.LookVector.X + r.Position.X, r.Position.Y, cam.LookVector.Z + r.Position.Z))
		end
	end)
	t4.m4Sk8Conn = m4RS.RenderStepped:Connect(function()
		local hh = m4Sk8GetHum()
		local c = m4LP.Character
		local r = c and c:FindFirstChild("HumanoidRootPart")
		if not hh or not r then return end
		pcall(function() hh.WalkSpeed = 60 hh.AutoRotate = false end)
		local lv = r.CFrame.LookVector
		local flat = Vector3.new(lv.X, 0, lv.Z)
		if flat.Magnitude > 0 then hh:Move(flat.Unit) else hh:Move(Vector3.new(0,0,0)) end

	if t4.m4Sk8Conn then t4.m4Sk8Conn:Disconnect() t4.m4Sk8Conn = nil end

	pcall(function() m4ACRemote:FireServer("UseActorAbility", { "Clone" }) end)

local function m4ACFacing(a, b)
	if not a or not b then return false end
	local d = b.Position - a.Position
	if d.Magnitude <= 0.001 then return true end
	return a.CFrame.LookVector:Dot(d.Unit) >= t1.m4AutoCloneDot
end

local function m4ACUnbind(model)
	local c = t4.m4AutoCloneHandlers[model]
	if c then c:Disconnect() t4.m4AutoCloneHandlers[model] = nil end
	t4.m4AutoCloneListeners[model] = nil
end

local function m4ACBind(model, animator)
	if not t1.m4AutoClone or not model or not model.Parent or not animator or not animator.Parent then return end
	if t4.m4AutoCloneHandlers[model] then return end
	t4.m4AutoCloneHandlers[model] = animator.AnimationPlayed:Connect(function(anim)
		if not t1.m4AutoClone then return end
		if not model.Parent then m4ACUnbind(model) return end
		local a = anim and anim.Animation
		local id = a and a.AnimationId
		if not id or not m4AutoCloneIds[id] then return end
		local char = m4LP.Character
		if not char or not char.Parent then return end
		local cr = char:FindFirstChild("HumanoidRootPart")
		local mr = model:FindFirstChild("HumanoidRootPart")
		if not cr or not mr then return end
		if (cr.Position - mr.Position).Magnitude > t1.m4AutoCloneRange then return end
		if t1.m4AutoCloneFacing and not m4ACFacing(mr, cr) then return end
		m4ACFire()
	end)
end

local function m4ACWatch(model)
	if not t1.m4AutoClone or not model:IsA("Model") or not model.Parent then return end
	if t4.m4AutoCloneListeners[model] then return end
	t4.m4AutoCloneListeners[model] = true
	task.spawn(function()
		while t1.m4AutoClone and model.Parent do
			local h = model:FindFirstChildOfClass("Humanoid")
			local an = h and h:FindFirstChildOfClass("Animator")
			if an then m4ACBind(model, an) return end
			task.wait(0.05)
		end
		if not t1.m4AutoClone or not model.Parent then m4ACUnbind(model) end
	end)
end

local function m4ACCleanup()
	for k in pairs(t4.m4AutoCloneListeners) do
		if not k or not k.Parent then m4ACUnbind(k) end
	end
end

local function m4ACStop()
	for k, c in pairs(t4.m4AutoCloneHandlers) do
		if c then c:Disconnect() end
		t4.m4AutoCloneHandlers[k] = nil
	end
	table.clear(t4.m4AutoCloneListeners)
end

local function m4ACStart()
	if t4.m4AutoCloneThread then return end
	t4.m4AutoCloneThread = task.spawn(function()
		while t1.m4AutoClone do
			local P = workspace:FindFirstChild("Players")
			local K = P and P:FindFirstChild("Killers")
			if K then
				for _, c in ipairs(K:GetChildren()) do
					if c:IsA("Model") then m4ACWatch(c) end
				end
			end
			m4ACCleanup()
			task.wait(0.05)
		end
		m4ACStop()
		t4.m4AutoCloneThread = nil
	end)
end

local m4BlockAnimState = {}

-- Auto Block (M4)
local function m4BlockStart()
	if t4.m4AutoBlockThread then return end
	t4.m4AutoBlockThread = task.spawn(function()
		while t1.m4AutoBlock do
			local char = m4LP.Character
			local myRoot = char and char:FindFirstChild("HumanoidRootPart")
			if myRoot then
				for _, killer in ipairs(m4KillersCache:GetChildren()) do
					if killer:IsA("Model") then
						local kr = killer:FindFirstChild("HumanoidRootPart")
						local inRange = kr and (myRoot.Position - kr.Position).Magnitude <= t1.m4AutoBlockRange
						if t1.m4AutoBlockType == "Sound" then
							for _, d in ipairs(killer:GetDescendants()) do
								if d:IsA("Sound") and m4BlockSoundIds[d.SoundId] then
									local fire = inRange
									if t1.m4AutoBlockFacing and inRange then
										local u = (myRoot.Position - kr.Position).Unit
										fire = kr.CFrame.LookVector:Dot(u) > t1.m4AutoBlockDot
									end
									if d.IsPlaying then
										if m4BlockSoundState[d] == nil then m4BlockSoundState[d] = not fire end
										if fire and not m4BlockSoundState[d] and not m4BlockAnimState[d] then
											m4BlockAnimState[d] = true
											m4NetworkRE:FireServer("UseActorAbility", { "Block" })
										end
									else
										m4BlockSoundState[d] = nil
										m4BlockAnimState[d] = nil
									end
								end
							end
						elseif t1.m4AutoBlockType == "Animation" then
							local h = killer:FindFirstChildOfClass("Humanoid")
							local an = h and h:FindFirstChildOfClass("Animator")
							if an then
								for _, tr in ipairs(an:GetPlayingAnimationTracks()) do
									local aid = tr.Animation and tr.Animation.AnimationId
									if aid and m4BlockAnimIds[aid] then
										local fire = inRange
										if t1.m4AutoBlockFacing and inRange then
											local u = (myRoot.Position - kr.Position).Unit
											fire = kr.CFrame.LookVector:Dot(u) > t1.m4AutoBlockDot
										end
										if m4BlockSoundState[tr] == nil then m4BlockSoundState[tr] = not fire end
										if fire and not m4BlockSoundState[tr] and not m4BlockAnimState[tr] then
											m4BlockAnimState[tr] = true
											m4NetworkRE:FireServer("UseActorAbility", { "Block" })
										end
									end
								end
								for k in pairs(m4BlockSoundState) do
									if typeof(k) == "Instance" and k:IsA("AnimationTrack") and not k.IsPlaying then
										m4BlockSoundState[k] = nil
										m4BlockAnimState[k] = nil
									end
								end
							end
						end
					end
				end
			end
			task.wait(0.1)
		end
		table.clear(m4BlockSoundState)
		table.clear(m4BlockAnimState)
		t4.m4AutoBlockThread = nil
	end)
end

-- Punch Aim (M4)
local function m4PunchAimStart()
	if t4.m4PunchAimThread then return end
	t4.m4PunchAimThread = task.spawn(function()
		while t1.m4PunchAim do
			local char = m4LP.Character
			local h = char and char:FindFirstChildOfClass("Humanoid")
			local root = char and char:FindFirstChild("HumanoidRootPart")
			if h and root then
				for _, track in ipairs(h:GetPlayingAnimationTracks()) do
					local a = track.Animation
					if a and m4PunchAimIds[a.AnimationId] then
						local best, bestD = nil, math.huge
						for _, k in ipairs(m4KillersCache:GetChildren()) do
							local kr = k:FindFirstChild("HumanoidRootPart")
							if kr then
								local d = (root.Position - kr.Position).Magnitude
								if d < bestD then bestD = d best = kr end
							end
						end
						if not best then break end
						local saved = h.AutoRotate
						h.AutoRotate = false
						local expiry = tick() + 0.5
						while tick() < expiry and t1.m4PunchAim do
							local parent = best.Parent
							local kh = parent and parent:FindFirstChildOfClass("Humanoid")
							if kh then
								local aim = best.Position + kh.MoveDirection * t1.m4PunchAimPredict
								root.CFrame = CFrame.lookAt(root.Position, Vector3.new(aim.X, root.Position.Y, aim.Z))
							end
							task.wait()
						end
						h.AutoRotate = saved
						break
					end
				end
			end
			task.wait(0.03)
		end
		t4.m4PunchAimThread = nil
	end)
end

local function m4M1Snap()
	if not t1.m4M1Aim then return end
	local char = m4LP.Character
	if not char then return end
	local root = char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local nearest, nd = nil, math.huge
	for _, p in ipairs(m4Players:GetPlayers()) do
		if p ~= m4LP then
			local c = p.Character
			local cr = c and c:FindFirstChild("HumanoidRootPart")
			if cr then
				local d = (root.Position - cr.Position).Magnitude
				if d < nd then nd = d nearest = cr end
			end
		end
	end
	if not nearest then return end
	local start = tick()
	local conn
	conn = m4RS.RenderStepped:Connect(function()
		if not t1.m4M1Aim or tick() - start >= 1 or not root.Parent or not nearest.Parent then
			conn:Disconnect()
			return
		end
		local aim = nearest.Position + nearest.CFrame.LookVector * 4
		root.CFrame = CFrame.lookAt(root.Position, Vector3.new(aim.X, root.Position.Y, aim.Z))
	end)
end

if getgenv().emergency_stop == nil then getgenv().emergency_stop = false end

local function m4StudsToPower(s) return s * 6 end

local m4HitboxExtender = {}
function m4HitboxExtender:ExtendHitbox(studs, duration)
	local power = m4StudsToPower(studs)
	local start = tick()
	if getgenv().emergency_stop then getgenv().emergency_stop = false end
	repeat
		m4RS.Heartbeat:Wait()
		while true do
			local c = m4LP.Character
			if c and c.Parent and c:FindFirstChild("HumanoidRootPart") and c.HumanoidRootPart.Parent then break end
			m4RS.Heartbeat:Wait()
		end
		local root = m4LP.Character.HumanoidRootPart
		local v = root.Velocity
		root.Velocity = v + root.CFrame.LookVector * power
		m4RS.RenderStepped:Wait()
		local c2 = m4LP.Character
		if c2 and c2.Parent and c2:FindFirstChild("HumanoidRootPart") then
			c2.HumanoidRootPart.Velocity = v
		end
	until tick() - start > tonumber(duration) or getgenv().emergency_stop
	if getgenv().emergency_stop then getgenv().emergency_stop = false end
end
function m4HitboxExtender:StopExtendingHitbox() getgenv().emergency_stop = true end

local function m4HitboxBind(btn)
	if not btn:IsA("ImageButton") then return end
	if not m4M1Buttons[btn.Name] then return end
	btn.Activated:Connect(function()
		if not t1.m4HitboxExpand then return end
		m4HitboxExtender:ExtendHitbox(t1.m4HitboxRange, t1.m4HitboxDur)
	end)
end
